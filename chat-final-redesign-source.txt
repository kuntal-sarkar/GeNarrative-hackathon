import IOSSelect from "../ui/IOSSelect";
import { useEffect, useRef, useState } from "react";
import { useLocation } from "react-router-dom";
import { aiConfigured, requestProjectAI, requestProjectAIWithAttachments } from "../../services/workspaceApi";
import { BookOpen, Check, Copy, Menu, MessageCircle, MoreHorizontal, Paperclip, Pencil, Plus, Send, ArrowUp, LoaderCircle, Trash2, X } from "lucide-react";
import "./ProjectAIChat.css";
import "./ProjectAIChatPolish.css";

const uid = () => crypto.randomUUID();
const ATTACHMENT_LIMIT = 5;
const MAX_FILE_BYTES = 10 * 1024 * 1024;
const MAX_TOTAL_BYTES = 25 * 1024 * 1024;
const ALLOWED_EXTENSIONS = new Set(["pdf", "txt", "md", "png", "jpg", "jpeg", "webp"]);
const formatSize = bytes => bytes < 1024 * 1024 ? `${Math.ceil(bytes / 1024)} KB` : `${(bytes / (1024 * 1024)).toFixed(1)} MB`;

const makeThread = (title = "New conversation", target = "project") => ({ id: uid(), title, target, messages: [], createdAt: new Date().toISOString() });
const normalize = (project) => {
  if (Array.isArray(project.chatThreads)) return project.chatThreads;
  const legacy = Array.isArray(project.chatMessages) ? project.chatMessages : [];
  return legacy.length ? [{ id: "legacy-chat", title: "Previous conversation", target: "project", messages: legacy }] : [];
};
export default function ProjectAIChat({ project, onSave, onOpenStory }) {
  const location = useLocation();
  const threads = normalize(project);
  const [activeId, setActiveId] = useState(null);
  const [input, setInput] = useState("");
  const [target, setTarget] = useState("project");
  const [busy, setBusy] = useState(false);
  const [copiedId, setCopiedId] = useState(null);
  const copyTimerRef = useRef(null);
  const textareaRef = useRef(null);
  const [error, setError] = useState("");
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [proposal, setProposal] = useState(null);
  const [revision, setRevision] = useState("");
  const [attachments, setAttachments] = useState([]);
  const attachmentsRef = useRef([]);
  const fileInputRef = useRef(null);
  const addFiles = fileList => {
    const incoming = Array.from(fileList || []);
    if (!incoming.length) return;
    const current = attachmentsRef.current;
    const next = [...current];
    for (const file of incoming) {
      const ext = file.name.split(".").pop()?.toLowerCase();
      if (!ALLOWED_EXTENSIONS.has(ext)) { setError(`Unsupported file: ${file.name}. Use PDF, TXT, MD, PNG, JPG or WebP.`); continue; }
      if (!file.size || file.size > MAX_FILE_BYTES) { setError(`${file.name}: files must be non-empty and no larger than 10 MB.`); continue; }
      if (next.length >= ATTACHMENT_LIMIT) { setError(`You can attach up to ${ATTACHMENT_LIMIT} files per message.`); break; }
      if (next.reduce((total, item) => total + item.file.size, 0) + file.size > MAX_TOTAL_BYTES) { setError("Total attachment size cannot exceed 25 MB."); break; }
      if (next.some(item => item.file.name === file.name && item.file.size === file.size && item.file.lastModified === file.lastModified)) continue;
      next.push({ id: uid(), file, previewUrl: file.type.startsWith("image/") ? URL.createObjectURL(file) : null });
    }
    attachmentsRef.current = next;
    setAttachments(next);
  };
  const clearAttachments = () => {
    attachmentsRef.current.forEach(item => { if (item.previewUrl) URL.revokeObjectURL(item.previewUrl); });
    attachmentsRef.current = [];
    setAttachments([]);
  };
  const removeAttachment = id => {
    const item = attachmentsRef.current.find(item => item.id === id);
    if (item?.previewUrl) URL.revokeObjectURL(item.previewUrl);
    const next = attachmentsRef.current.filter(item => item.id !== id);
    attachmentsRef.current = next;
    setAttachments(next);
  };
  useEffect(() => () => { attachmentsRef.current.forEach(item => { if (item.previewUrl) URL.revokeObjectURL(item.previewUrl); }); }, []);

  const copyMessage = async (id, content) => {
    try {
      await navigator.clipboard.writeText(content);
      setCopiedId(id);
      window.clearTimeout(copyTimerRef.current);
      copyTimerRef.current = window.setTimeout(() => setCopiedId(null), 1800);
    } catch {
      setError("Couldn't copy message. Check browser clipboard permissions.");
    }
  };
  useEffect(() => () => window.clearTimeout(copyTimerRef.current), []);
  const resizeInput = (el) => {
    if (!el) return;
    el.style.height = "auto";
    el.style.height = `${Math.min(el.scrollHeight, 156)}px`;
  };
  const scrollRef = useRef(null);
  const thread = threads.find(t=>t.id===activeId) || threads.find(t=>t.id===project.activeChatThreadId) || threads[0] || null;
  useEffect(()=>{ setActiveId(null); setTarget("project"); setProposal(null); setInput(""); clearAttachments(); },[project.id]);
  useEffect(()=>{ if (location.state?.chatIntent === "story") setTarget("story"); },[location.state]);
  useEffect(()=>{scrollRef.current?.scrollTo({top: scrollRef.current.scrollHeight,behavior:"smooth"});},[thread?.messages?.length]);
  function commit(nextOrUpdater, id) {
    onSave(latest => {
      if (!latest) return latest;
      const all = normalize(latest);
      const next = typeof nextOrUpdater === "function" ? nextOrUpdater(all) : nextOrUpdater;
      return { ...latest, chatThreads: next, activeChatThreadId: id };
    });
    setActiveId(id || null);
  }
  function create() { const t=makeThread("New conversation",target); commit(all => [t,...all],t.id); setInput("");setProposal(null);setSidebarOpen(false); }
  function rename(id) { const old=threads.find(t=>t.id===id); const title=window.prompt("Conversation name",old?.title||"")?.trim(); if(title)commit(all => all.map(t=>t.id===id?{...t,title:title.slice(0,90)}:t), id); }
  function remove(id) {if(!window.confirm("Delete this conversation from the project draft? Save Changes to confirm."))return;const remaining=threads.filter(t=>t.id!==id);commit(all => all.filter(t=>t.id!==id),remaining[0]?.id||null);setActiveId(remaining[0]?.id||null);}
  function switchTo(id){setActiveId(id);setProposal(null);setError("");setSidebarOpen(false);}
  async function send(e) {
    e.preventDefault(); const content=input.trim(); if(!content||busy)return;
    if(!aiConfigured){setError("Flask AI backend isn't connected yet. No simulated replies are generated.");return;}
    const selectedFiles = attachmentsRef.current.map(item => item.file);
    const selectedMetadata = selectedFiles.map(file => ({ name: file.name, size: file.size, type: file.type || "application/octet-stream" }));
    const current=thread||makeThread(content.slice(0,55),target);
    const history=Array.isArray(current.messages)?current.messages:[];
    const userMessage={id:uid(),role:"user",content,...(selectedMetadata.length ? {attachments:selectedMetadata} : {}),sceneId:target.startsWith("scene:")?target.slice(6):null,at:new Date().toISOString()};
    const waiting=[...history,userMessage];
    // Append to the latest draft, not to the project snapshot captured before the request.
    commit(all => {
      const exists = all.some(t => t.id === current.id);
      const base = exists ? all : [current, ...all];
      return base.map(t => t.id === current.id ? {
        ...t, title: history.length ? t.title : content.slice(0, 60),
        messages: [...(t.messages || []), userMessage], target
      } : t);
    }, current.id);
    setInput("");setError("");setBusy(true);setProposal(null);
    if (textareaRef.current) textareaRef.current.style.height = "auto";
    try {
      const payload={
        message:content, target, sceneId:target.startsWith("scene:")?target.slice(6):null,
        language:project.language||"English", history:history.map(m=>({role:m.role,content:m.content})).slice(-24),
        context:{title:project.title,idea:project.idea,story:project.story,scenes:project.scenes,creativeDecisions:project.creativeDecisions}
      };
      const data=selectedFiles.length
        ? await requestProjectAIWithAttachments(project.id,"chat",payload,selectedFiles)
        : await requestProjectAI(project.id,"chat",payload);
      const answer=typeof data?.answer==="string"?data.answer.trim():"";
      if(!answer)throw new Error("Backend returned no answer.");
      const response={id:uid(),role:"assistant",content:answer,at:new Date().toISOString()};
      // Functional update preserves edits, other chats, renames and scene changes
      // made while the backend request was in flight.
      onSave(latest => ({ ...latest, chatThreads: normalize(latest).map(t =>
        t.id === current.id ? { ...t, messages: [...(t.messages || []), response] } : t
      ) }));
      clearAttachments();
      if(data?.proposedRevision && typeof data.proposedRevision.text==="string" && ["story","scene"].includes(data.proposedRevision.target)) {
        setProposal(data.proposedRevision); setRevision(data.proposedRevision.text);
      }
    } catch(err) {
      // A failed attachment upload must never be represented as a successfully sent file.
      if (selectedFiles.length) {
        onSave(latest => ({ ...latest, chatThreads: normalize(latest).map(t =>
          t.id === current.id ? { ...t, messages: (t.messages || []).filter(m => m.id !== userMessage.id) } : t
        ) }));
        setInput(content);
      }
      setError(err.message || "Request failed. Files remain attached so you can retry.");
    }
    finally {setBusy(false);}
  }
  function applyProposal(){
    if(!proposal||!revision.trim())return;
    if(proposal.target==="story") onSave(latest => ({ ...latest, story:revision }));
    else if(proposal.target==="scene" && proposal.sceneId) {
      const match=(project.scenes||[]).some(s=>s.id===proposal.sceneId);
      if(!match){setError("The referenced scene no longer exists.");return;}
      onSave(latest => ({ ...latest, scenes:(latest.scenes||[]).map(s=>s.id===proposal.sceneId?{...s,description:revision}:s) }));
    } else return;
    setProposal(null);setRevision("");setError("Revision added to unsaved project draft. Click Save Changes at the top to keep it.");
  }
  return <section className="ff-chatroom" aria-label="Project AI writing room">
    <aside className={`ff-chat-sidebar ${sidebarOpen?"open":""}`}>
      <div className="ff-chat-sidebar-head"><strong>Conversations</strong><button type="button" title="Close chat list" className="ff-chat-mobile-close" onClick={()=>setSidebarOpen(false)}><X size={17}/></button></div>
      <button className="ff-chat-new" onClick={create}><Plus size={17}/> New chat</button>
      <div className="ff-thread-list">{threads.length?threads.map(t=><div key={t.id} className={`ff-thread-row ${t.id===thread?.id?"selected":""}`}><button className="ff-thread-open" type="button" onClick={()=>switchTo(t.id)}><MessageCircle size={15}/><span>{t.title}</span></button><button title="Rename" onClick={()=>rename(t.id)}><Pencil size={14}/></button><button title="Delete" onClick={()=>remove(t.id)}><Trash2 size={14}/></button></div>):<p className="ff-chat-muted">No conversations yet. Start one whenever you're ready.</p>}</div>
      <div className="ff-chat-sidebar-foot">Conversations are stored with this project after Save Changes.</div>
    </aside>
    <div className="ff-chat-center">
      <header className="ff-chat-header"><button className="ff-chat-menu" onClick={()=>setSidebarOpen(true)} aria-label="Open conversations"><Menu size={19}/></button><div><span>PROJECT WRITING ROOM</span><strong>{thread?.title||"New conversation"}</strong></div><button className="ff-chat-storylink" type="button" onClick={onOpenStory}><BookOpen size={16}/> Story editor</button></header>
      <div className="ff-chat-target"><label>AI context<IOSSelect value={target} onChange={e=>setTarget(e.target.value)}><option value="project">Entire project</option><option value="story">Full story / rewrite</option>{(project.scenes||[]).map((s,i)=><option key={s.id} value={`scene:${s.id}`}>Scene {i+1}: {s.name||s.heading||"Untitled"}</option>)}</IOSSelect></label><span>{aiConfigured?"Backend configured":"AI backend not connected"}</span></div>
      <div className="ff-chat-transcript" ref={scrollRef} role="log" aria-live="polite">{thread?.messages?.length?thread.messages.map(m=><article key={m.id} className={`ff-chat-bubble ${m.role==="user"?"is-user":"is-ai"}`}><div className="ff-chat-who">{m.role==="user"?"You":"AI Co-Director"}<button type="button" className={`ff-copy-action ${copiedId === m.id ? "is-copied" : ""}`} title={copiedId === m.id ? "Copied" : "Copy message"} aria-label={copiedId === m.id ? "Copied to clipboard" : "Copy message"} onClick={() => copyMessage(m.id, m.content)}>{copiedId === m.id ? <Check size={15}/> : <Copy size={15}/>}</button></div><p>{m.content}</p>{Array.isArray(m.attachments) && m.attachments.length > 0 && <div className="ff-chat-attachment-history" aria-label="Files sent with this message">{m.attachments.map((item,i)=><span key={`${item.name}-${i}`}><Paperclip size={12}/>{item.name}</span>)}</div>}</article>):<div className="ff-chat-welcome"><div className="ff-chat-welcome-icon"><MessageCircle size={27}/></div><h2>Your creative thinking partner</h2><p>Discuss plot, character motives, screenplay rewrites or any specific scene. Create multiple chats, continue past discussions and approve proposed edits before they reach your story.</p>{!aiConfigured&&<p className="ff-chat-offline">Connect the Flask + Gemini backend to send messages. No fake AI content is generated.</p>}</div>}{busy && <article className="ff-chat-thinking" role="status" aria-live="polite"><span className="ff-thinking-dots" aria-hidden="true"><i/><i/><i/></span><span>AI Co-Director is shaping a response...</span></article>}</div>
      {proposal&&<div className="ff-chat-proposal"><div className="ff-chat-proposal-head"><strong>Proposed {proposal.target} revision â€” review before applying</strong><button onClick={()=>setProposal(null)}><X size={16}/></button></div><textarea value={revision} onChange={e=>setRevision(e.target.value)} rows={7} aria-label="Edit proposed story text"/><div><button className="ff-chat-apply" onClick={applyProposal}><Check size={17}/> Apply to draft</button><small>Not saved until you click Save Changes above.</small></div></div>}
      {error&&<p className="ff-chat-error" role="status">{error}</p>}
      <form onSubmit={send} className="ff-chat-composer">
        <input ref={fileInputRef} className="ff-chat-file-input" type="file" accept=".pdf,.txt,.md,.png,.jpg,.jpeg,.webp" multiple onChange={e => { addFiles(e.target.files); e.target.value=""; }} aria-label="Attach files" />
        {attachments.length > 0 && <div className="ff-chat-attachments" aria-label="Selected attachments">{attachments.map(item => <div className="ff-chat-attachment" key={item.id}>{item.previewUrl ? <img src={item.previewUrl} alt="" /> : <Paperclip size={17}/>}<span title={item.file.name}>{item.file.name}<small>{formatSize(item.file.size)}</small></span><button type="button" aria-label={`Remove ${item.file.name}`} title="Remove attachment" onClick={() => removeAttachment(item.id)} disabled={busy}><X size={15}/></button></div>)}</div>}
        <div className="ff-chat-compose-shell">
          <button className="ff-chat-plus" type="button" title="Attach a file" aria-label="Attach a file" onClick={() => fileInputRef.current?.click()} disabled={busy}><Plus size={21}/></button>
          <textarea ref={textareaRef} rows={1} placeholder="Ask your AI Co-Director anything..." aria-label="Write a message" value={input} onChange={e => { setInput(e.target.value); resizeInput(e.currentTarget); }} onKeyDown={e => { if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) { e.preventDefault(); send(e); } }}/>
          <button className={`ff-chat-send-arrow ${busy ? "is-thinking" : ""}`} type="submit" aria-label={busy ? "AI is thinking" : "Send message"} title={busy ? "AI is thinking" : "Send message"} disabled={!input.trim() || busy || !aiConfigured}>{busy ? <LoaderCircle size={19}/> : <ArrowUp size={20}/>}</button>
        </div>
        <div className="ff-chat-compose-hint"><span>{busy ? "Crafting your next scene of thought..." : "Enter to send Â· Shift+Enter for a new line"}</span><span>Creative decisions stay yours</span></div>
      </form>
    </div>
  </section>;
}

export default function ProjectWorkspace() {
  const { projectId, section = "overview" } = useParams();

  const [project, setProject] = useState(() => getProject(projectId));
  const [dirty, setDirty] = useState(false);
  const [storyMode, setStoryMode] = useState("manual");
  const [status, setStatus] = useState("");
  const [error, setError] = useState("");
  const [mobileToolsOpen, setMobileToolsOpen] = useState(false);
  const [savedRevision, setSavedRevision] = useState(0);
  const [historyRevision, setHistoryRevision] = useState(0);
  const [showLeaveDialog, setShowLeaveDialog] = useState(false);
  const workspaceRef = useRef(null);
  const leaveDialogRef = useRef(null);
  // Check each workspace section independently so a draft in Story still shows
  // its indicator while the user works in Scenes or Shots.
  const storedSnapshot = savedRevision >= 0 ? getProject(projectId) : null;
  const draftSections = (() => {
    if (!project || !storedSnapshot) return [];
    const outstanding = [];
    if (!sameData(project.characters || [], storedSnapshot.characters || [])) outstanding.push("characters");
    if (["title", "idea", "genre", "story", "sourceFilename"].some(k => !sameData(project[k], storedSnapshot[k]))) outstanding.push("story");
    const dScenes = project.scenes || [], sScenes = storedSnapshot.scenes || [];
    if (!sameData(dScenes.map(({id,name,heading,description})=>({id,name,heading,description})), sScenes.map(({id,name,heading,description})=>({id,name,heading,description})))) outstanding.push("scenes");
    const extractShots = scenes => scenes.map(s => ({sceneId:s.id, shots: (s.shots || []).map(shot=>shot)}));
    if (!sameData(extractShots(dScenes), extractShots(sScenes))) outstanding.push("shots");
    for (const [key,fields] of Object.entries(SECTION_FIELDS)) {
      if (fields.some(field => !sameData(project[field],storedSnapshot[field]))) outstanding.push(key);
    }
    return outstanding;
  })();
  const hasUnsaved = draftSections.length > 0;
  const blocker = useBlocker(({ currentLocation, nextLocation }) => {
    if (!hasUnsaved) return false;
    const here = currentLocation.pathname.match(/^\/projects\/([^/]+)/);
    const there = nextLocation.pathname.match(/^\/projects\/([^/]+)/);
    return !here || !there || here[1] !== there[1];
  });
  useEffect(() => {
    setShowLeaveDialog(blocker.state === "blocked");
  }, [blocker.state]);
  function stayInProject() { setShowLeaveDialog(false); if (blocker.state === "blocked") blocker.reset(); }
  function leaveProject() { setShowLeaveDialog(false); if (blocker.state === "blocked") blocker.proceed(); }
  useWorkspaceKeyboard({ workspaceRef, leaveDialogRef, showLeaveDialog, stayInProject });
  function reviewDrafts() {
    const first = draftSections[0];
    stayInProject();
    if (first) navigate(`/projects/${encodeURIComponent(projectId)}/${first}`);
  }

  useEffect(() => {
    setProject(getProject(projectId));
    setDirty(false);
    setError("");
    setStatus("");
  }, [projectId]);

  useEffect(() => {
    const warn = (e) => { if (hasUnsaved) { e.preventDefault(); e.returnValue = ""; } };
    window.addEventListener("beforeunload", warn);
    return () => window.removeEventListener("beforeunload", warn);
  }, [hasUnsaved]);

  // Track the last workspace tool without changing existing project storage.
  useEffect(() => {
    if (!projectId || section === "overview" || !SECTIONS.some(item => item.key === section)) return;
    try { localStorage.setItem(lastSectionKey(projectId), section); } catch { /* Restricted browser storage. */ }
  }, [projectId, section]);

  const base = `/projects/${encodeURIComponent(projectId)}`;
  const current = SECTIONS.find(s => s.key === section);
  const CurrentIcon = current?.icon || LayoutDashboard;

  // Edits are held only in React state. The database/browser is not updated until Save Changes.
  function persist(nextOrUpdater) {
    setProject(previous => typeof nextOrUpdater === "function" ? nextOrUpdater(previous) : nextOrUpdater);
    setDirty(true);
    setStatus("Unsaved changes");
    setError("");
  }
  function update(field, value) { persist(previous => ({ ...previous, [field]: value })); }
  // An individual Save commits ONLY the selected fields/card. Every other draft stays in memory.
  function savePartial(merge, success = "Section saved") {
    try {
      const latest = getProject(projectId);
      if (!latest) throw new Error("Project was removed from storage.");
      const next = merge(latest);
      if (!next?.title?.trim()) throw new Error("Project title cannot be empty.");
      saveProject(next);
      const historyResult = recordVersion(projectId, latest, next, success);
      if (historyResult.recorded) setHistoryRevision(n => n + 1);
      setSavedRevision(n => n + 1);
      setStatus(historyResult.warning ? `${success}. ${historyResult.warning}` : success);
      setError("");
      // Do not overwrite any unsaved inputs in React state.
      const draftComparable = { ...project, updatedAt: null };
      const savedComparable = { ...next, updatedAt: null };
      setDirty(JSON.stringify(draftComparable) !== JSON.stringify(savedComparable));
    } catch (err) { setError(err?.message || "Could not save. Check available browser storage."); }
  }
  function restoreHistoryVersion(version, side) {
    if (hasUnsaved) { setError("Save or discard unsaved changes before restoring a version."); return; }
    try {
      const latest = getProject(projectId);
      if (!latest) throw new Error("Project was removed from storage.");
      const restored = applyVersionFields(latest, version, side);
      // Save current data FIRST so it stays recoverable, then update the workspace.
      const changed = JSON.stringify({ ...latest, updatedAt: null }) !== JSON.stringify({ ...restored, updatedAt: null });
      if (!changed) { setStatus("This version already matches the current saved fields."); setError(""); return; }
      saveProject(restored);
      const historyResult = recordVersion(projectId, latest, restored, `Restored ${version.label}`);
      setProject(getProject(projectId));
      setDirty(false);
      setSavedRevision(n => n + 1);
      if (historyResult.recorded) setHistoryRevision(n => n + 1);
      setStatus(historyResult.warning ? `Version restored. ${historyResult.warning}` : "Version restored successfully");
      setError("");
    } catch (err) { setError(err?.message || "Unable to restore the selected version."); }
  }
  function saveStoryDetails() {
    const { title, idea, genre } = project;
    savePartial(latest => ({ ...latest, title: title.trim(), idea, genre }), "Story details saved");
  }
  const savedProject = storedSnapshot;
  const storyDetailsSaved = Boolean(savedProject && ["title", "idea", "genre"].every(k => (savedProject[k] || "") === (project?.[k] || "")));
  const storyTextSaved = Boolean(savedProject && ["story", "sourceFilename"].every(k => (savedProject[k] || "") === (project?.[k] || "")));
  function saveStoryText() {
    const { story, sourceFilename } = project;
    savePartial(latest => ({ ...latest, story, sourceFilename }), "Story text saved");
  }
  function saveScene(sceneId) {
    const draftScene = project.scenes?.find(scene => scene.id === sceneId);
    if (!draftScene) return;
    savePartial(latest => {
      const original = [...(latest.scenes || [])];
      const pos = original.findIndex(scene => scene.id === sceneId);
      const savedScene = pos >= 0 ? { ...original[pos], name: draftScene.name, heading: draftScene.heading, description: draftScene.description } : { ...draftScene, shots: [] };
      if (pos >= 0) original[pos] = savedScene;
      else original.push(savedScene);
      return { ...latest, scenes: original.map((scene, i) => ({ ...scene, number: i + 1 })) };
    }, "Scene saved");
  }
  function saveShot(sceneId, shotId) {
    const draftScene = project.scenes?.find(scene => scene.id === sceneId);
    const draftShot = draftScene?.shots?.find(shot => shot.id === shotId);
    if (!draftShot) return;
    savePartial(latest => {
      const scenes = [...(latest.scenes || [])];
      let pos = scenes.findIndex(scene => scene.id === sceneId);
      if (pos < 0) throw new Error("Save this scene first, then save its shots.");
      const shots = [...(scenes[pos].shots || [])];
      const shotPos = shots.findIndex(shot => shot.id === shotId);
      if (shotPos < 0) shots.push(draftShot);
      else shots[shotPos] = draftShot;
      const draftOrder = (draftScene.shots || []).map(shot => shot.id);
      shots.sort((a,b) => draftOrder.indexOf(a.id)-draftOrder.indexOf(b.id));
      scenes[pos] = { ...scenes[pos], shots };
      return { ...latest, scenes };
    }, "Shot saved");
  }
  function isSceneSaved(scene) {
    const original = getProject(projectId)?.scenes?.find(s => s.id === scene.id);
    return Boolean(original && ["name", "heading", "description"].every(k => (original[k] || "") === (scene[k] || "")));
  }
  function isShotSaved(sceneId, shot) {
    const original = getProject(projectId)?.scenes?.find(s => s.id === sceneId)?.shots?.find(s => s.id === shot.id);
    return Boolean(original && ["name", "type", "angle", "movement", "lighting", "mood", "duration", "description", "notes"].every(k => (original[k] || "") == (shot[k] || "")));
  }
  function handleScenes(action, payload = {}) {
    let scenes = [...(project.scenes || [])];
    if (action === "add") scenes.push({ id: crypto.randomUUID(), name: "", heading: "", description: "", shots: [] });
    else if (action === "edit") scenes = scenes.map(s => s.id === payload.id && ["name", "heading", "description"].includes(payload.field) ? { ...s, [payload.field]: payload.value } : s);
    else if (action === "delete") {
      const target = scenes.find(s => s.id === payload.id);
      if (target?.shots?.length && !window.confirm("This scene has shots. Delete the scene and all its shots?")) return;
      scenes = scenes.filter(s => s.id !== payload.id);
      if (getProject(projectId)?.scenes?.some(s => s.id === payload.id)) {
        savePartial(latest => ({ ...latest, scenes: (latest.scenes || []).filter(s => s.id !== payload.id).map((s,i) => ({ ...s, number:i+1 })) }), "Scene deleted");
      }
    } else return;
    if (action === "delete") {
      setProject(previous => ({ ...previous, scenes: scenes.map((s, i) => ({ ...s, number: i + 1 })) }));
      setDirty(JSON.stringify({ ...project, scenes: scenes.map((s,i) => ({...s,number:i+1})), updatedAt:null }) !== JSON.stringify({ ...getProject(projectId), updatedAt:null }));
    } else persist(previous => ({ ...previous, scenes: scenes.map((s, i) => ({ ...s, number: i + 1 })) }));
  }

  function handleShots(sceneId, action, payload = {}) {
    const scene = project.scenes?.find(s => s.id === sceneId);
    if (!scene) return;
    const currentShots = [...(scene.shots || [])];
    const isSaved = shotId => Boolean(getProject(projectId)?.scenes?.find(s => s.id === sceneId)?.shots?.some(s => s.id === shotId));
    let nextShots = currentShots;
    if (action === "add") nextShots = [...currentShots, { id: crypto.randomUUID(), name: "", type: "", angle: "", movement: "", lighting: "", mood: "", duration: "", description: "", notes: "" }];
    else if (action === "edit") {
      if (!["name", "type", "angle", "movement", "lighting", "mood", "duration", "description", "notes"].includes(payload.field)) return;
      nextShots = currentShots.map(shot => shot.id === payload.id ? { ...shot, [payload.field]: payload.value } : shot);
    } else if (action === "delete") {
      if (!window.confirm("Delete this shot permanently?")) return;
      nextShots = currentShots.filter(shot => shot.id !== payload.id);
      if (isSaved(payload.id)) savePartial(latest => ({ ...latest, scenes: (latest.scenes || []).map(s => s.id !== sceneId ? s : ({ ...s, shots: (s.shots || []).filter(shot => shot.id !== payload.id) })) }), "Shot deleted");
    } else if (action === "move") {
      const i = currentShots.findIndex(shot => shot.id === payload.id);
      const j = i + payload.direction;
      if (i < 0 || j < 0 || j >= currentShots.length) return;
      nextShots = [...currentShots];
      [nextShots[i], nextShots[j]] = [nextShots[j], nextShots[i]];
      const savedOrder = nextShots.map(shot => shot.id);
      // Saved items can reorder immediately; unsaved shots retain draft-only order.
      if (currentShots.every(shot => isSaved(shot.id))) {
        savePartial(latest => ({ ...latest, scenes: (latest.scenes || []).map(s => s.id !== sceneId ? s : ({ ...s,
          shots: [...(s.shots || [])].sort((a,b) => savedOrder.indexOf(a.id)-savedOrder.indexOf(b.id)) })) }), "Shot order saved");
      } else {
        setStatus("Shot moved in draft. Save new shots to keep their order.");
      }
    } else return;
    const nextProject = { ...project, scenes: (project.scenes || []).map(s => s.id === sceneId ? ({ ...s, shots: nextShots }) : s) };
    setProject(nextProject);
    const stored = getProject(projectId);
    setDirty(JSON.stringify({ ...nextProject, updatedAt:null }) !== JSON.stringify({ ...stored, updatedAt:null }));
    setError("");
  }

  async function uploadScript(file) {
    if (!file) return;

    try {
      const imported = await readScript(file);
      // Append rather than replace, including when edits happen while PDF extraction is running.
      persist(previous => ({ ...previous,
        story: [previous.story?.trimEnd(), imported.text.trim()].filter(Boolean).join("\n\n"),
        sourceFilename: imported.filename,
      }));
      setStatus(`${imported.filename} appended below the existing story. Save Story Text to keep it.`);
    } catch (err) { setError(err.message || "Could not read script."); }
  }

  const memoryCount = project.creativeDecisions?.length || 0;
